(defrule macbookair-iphone16-contado
 (orden (clienteid ?cte) (prodid c01) (metodopagoid contado))
 (orden (clienteid ?cte) (prodid sp01) (metodopagoid contado))
 (cliente (clienteid ?cte) (nombre ?name))
 (vales (cteid ?cte) (monto ?cant))
 (computadora (id c01) (precio ?cprice))
 (smartphone (id sp01) (precio ?spprice))
 =>
 (bind ?total (+ ?cprice ?spprice))
 (bind ?vales (* (/ ?total 1000) 100))
 (bind ?valestotales (+ ?cant ?vales))
 (printout t ?name " ha sumado a sus vales la cantidad de $" ?vales " por comprar el combo apple de contado" crlf)
 (assert (vales (cteid ?cte) (monto ?valestotales)))
)

(defrule descvales-smartphones
 (orden (clienteid ?cte) (prodid ?prod) (categoria smartphone) (cantidad ?cant) (metodopagoid vales))
 (cliente (clienteid ?cte) (nombre ?name))
 (vales (cteid ?cte) (monto ?vales))
 (smartphone (id ?prod) (precio ?price))
 =>
 (bind ?desc (* ?cant ?price))
 (bind ?valesfinales (- ?vales ?desc))
 (assert (vales (cteid ?cte) (monto ?valesfinales)))
 (printout t ?name " después de esta compra le quedan $" ?valesfinales " en vales" crlf)
)
